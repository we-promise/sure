# Transaction Location Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let any transaction optionally carry a place (name + coordinates), and add a preview-gated "Map" page that plots every located transaction as a pin.

**Architecture:** Three nullable columns on `entries` hold the location. A server-side `Geocoding::NominatimClient` wraps OpenStreetMap Nominatim (forward + reverse geocoding, cached, rate-limit-respecting) behind a thin `LocationSearchesController` proxy. The existing `TransactionsController#update` gains three new permitted params — no new write endpoint. A new `MapsController#index` serves the located entries as JSON to a Leaflet + Leaflet.markercluster map, rendered by a Stimulus controller. Both the Map page and the location picker on the transaction form are gated behind the existing `PreviewGateable` per-user toggle.

**Tech Stack:** Rails 8.1, Minitest + VCR + Mocha, ViewComponent (`DS::*`), Stimulus, importmap-rails, Leaflet (vendored via `bin/importmap pin`), OpenStreetMap Nominatim (no API key).

**Spec:** `docs/superpowers/specs/2026-11-22-transaction-location-map-design.md`

```ruby
# app/models/geocoding/nominatim_client.rb
require "net/http"
require "json"

class Geocoding::NominatimClient
  BASE_URL = "https://nominatim.openstreetmap.org"
  USER_AGENT = "Sure (self-hosted personal finance app; https://github.com/we-promise/sure)"
  CACHE_EXPIRY = 1.day

  Result = Struct.new(:lat, :lng, :display_name, keyword_init: true)

  class RequestFailed < StandardError; end

  # Forward geocoding: free-text query -> up to 5 candidate places.
  def search(query)
    return [] if query.blank?

    Rails.cache.fetch(cache_key("search", query), expires_in: CACHE_EXPIRY) do
      response = get("/search", q: query, format: "json", limit: 5)
      Array(response).map { |place| to_result(place) }
    end
  end

  # Reverse geocoding: coordinates -> the single best display name.
  def reverse(lat, lng)
    Rails.cache.fetch(cache_key("reverse", "#{lat},#{lng}"), expires_in: CACHE_EXPIRY) do
      response = get("/reverse", lat: lat, lon: lng, format: "json")
      response.present? ? to_result(response) : nil
    end
  end

  private

    def to_result(place)
      Result.new(
        lat: place["lat"].to_f,
        lng: place["lon"].to_f,
        display_name: place["display_name"]
      )
    end

    def cache_key(action, value)
      "geocoding/nominatim/#{action}/#{value}"
    end

    def get(path, params)
      uri = URI("#{BASE_URL}#{path}")
      uri.query = URI.encode_www_form(params)

      request = Net::HTTP::Get.new(uri)
      request["User-Agent"] = USER_AGENT

      response = Net::HTTP.start(uri.host, uri.port, use_ssl: true) { |http| http.request(request) }

      raise RequestFailed, "Nominatim returned #{response.code}" unless response.is_a?(Net::HTTPSuccess)

      JSON.parse(response.body)
    end
end
```

```ruby
# app/controllers/location_searches_controller.rb
class LocationSearchesController < ApplicationController
  before_action :require_preview_features!

  def index
    results = Geocoding::NominatimClient.new.search(params[:q])

    render json: results.map { |r| { lat: r.lat, lng: r.lng, display_name: r.display_name } }
  rescue Geocoding::NominatimClient::RequestFailed, Net::OpenTimeout, Net::ReadTimeout
    render json: [], status: :service_unavailable
  end
end
```

The rescue is deliberately scoped to this action rather than handled inside `NominatimClient` itself: the client's job is to report failure accurately (raise), and each caller decides what degrading gracefully looks like for it — here, an empty result set with a non-2xx status so the location picker's JS can tell "no matches" apart from "search is down" (per the spec's error-handling section).

```ruby
# app/controllers/maps_controller.rb
class MapsController < ApplicationController
  before_action :require_preview_features!

  def show
    @located_entries = filtered_located_entries.includes(:account, :entryable)

    @map_data = @located_entries.map do |entry|
      {
        id: entry.id,
        lat: entry.location_lat.to_f,
        lng: entry.location_lng.to_f,
        name: entry.location_name,
        transaction_name: entry.name,
        amount: entry.amount_money.format,
        date: entry.date.to_s,
        transaction_path: transaction_path(entry)
      }
    end

    @accounts = Current.family.accounts.accessible_by(Current.user).alphabetically
  end

  private

    def filtered_located_entries
      scope = Current.family.entries
        .located
        .joins(:account)
        .merge(Account.accessible_by(Current.user))

      scope = scope.where(account_id: params[:account_id]) if params[:account_id].present?
      scope = scope.where("entries.date >= ?", params[:start_date]) if params[:start_date].present?
      scope = scope.where("entries.date <= ?", params[:end_date]) if params[:end_date].present?
      scope
    end
end
```

```ruby
      entry_params[:location_name] = nil if entry_params[:location_name].blank?

      if nature.present? && entry_params[:amount].present?
        signed_amount = nature == "inflow" ? -entry_params[:amount].to_d : entry_params[:amount].to_d
        entry_params = entry_params.merge(amount: signed_amount)
      end

      entry_params
    end
```

```javascript
// app/javascript/controllers/location_picker_controller.js
import { Controller } from "@hotwired/stimulus";
import L from "leaflet";

// Connects to data-controller="location-picker"
export default class extends Controller {
  static targets = [
    "lat", "lng", "name",
    "display", "displayText",
    "search", "searchInput", "suggestions",
    "confirmMap",
  ];

  connect() {
    this.marker = null;
    this.confirmMapInstance = null;
  }

  disconnect() {
    this.confirmMapInstance?.remove();
  }

  search(event) {
    const query = event.target.value.trim();

    clearTimeout(this.debounceTimeout);
    if (query.length < 3) {
      this.suggestionsTarget.innerHTML = "";
      return;
    }

    this.debounceTimeout = setTimeout(() => this.#fetchSuggestions(query), 300);
  }

  select(event) {
    const { lat, lng, name } = event.params;
    this.#showConfirmMap(lat, lng, name);
  }

  clear() {
    this.latTarget.value = "";
    this.lngTarget.value = "";
    this.nameTarget.value = "";
    this.displayTarget.classList.add("hidden");
    this.searchTarget.classList.remove("hidden");
    this.confirmMapTarget.classList.add("hidden");
    this.confirmMapInstance?.remove();
    this.confirmMapInstance = null;
    this.element.closest("form").requestSubmit();
  }

  async #fetchSuggestions(query) {
    const response = await fetch(`/location_searches?q=${encodeURIComponent(query)}`, {
      headers: { Accept: "application/json" },
    });

    if (!response.ok) {
      this.suggestionsTarget.innerHTML = `<li>${this.#t("search_unavailable")}</li>`;
      return;
    }

    const results = await response.json();

    if (results.length === 0) {
      this.suggestionsTarget.innerHTML = `<li>${this.#t("no_matches")}</li>`;
      return;
    }

    this.suggestionsTarget.innerHTML = "";
    results.forEach((result) => {
      const item = document.createElement("li");
      item.textContent = result.display_name;
      item.dataset.action = "click->location-picker#select";
      item.dataset.locationPickerLatParam = result.lat;
      item.dataset.locationPickerLngParam = result.lng;
      item.dataset.locationPickerNameParam = result.display_name;
      this.suggestionsTarget.appendChild(item);
    });
  }

  #showConfirmMap(lat, lng, name) {
    this.latTarget.value = lat;
    this.lngTarget.value = lng;
    this.nameTarget.value = name;
    this.displayTextTarget.textContent = name;

    this.searchTarget.classList.add("hidden");
    this.displayTarget.classList.remove("hidden");
    this.confirmMapTarget.classList.remove("hidden");

    this.confirmMapInstance?.remove();
    this.confirmMapInstance = L.map(this.confirmMapTarget).setView([lat, lng], 15);
    L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
      attribution: "&copy; OpenStreetMap contributors",
    }).addTo(this.confirmMapInstance);

    this.marker = L.marker([lat, lng], { draggable: true }).addTo(this.confirmMapInstance);
    this.marker.on("dragend", () => this.#onDrag());

    this.element.closest("form").requestSubmit();
  }

  async #onDrag() {
    const { lat, lng } = this.marker.getLatLng();
    this.latTarget.value = lat;
    this.lngTarget.value = lng;

    const response = await fetch(`/location_searches/reverse?lat=${lat}&lng=${lng}`, {
      headers: { Accept: "application/json" },
    });

    if (response.ok) {
      const result = await response.json();
      this.nameTarget.value = result.display_name;
      this.displayTextTarget.textContent = result.display_name;
    }

    this.element.closest("form").requestSubmit();
  }

  #t(key) {
    const messages = {
      search_unavailable: "Search unavailable, try again",
      no_matches: "No matches, try a different search",
    };
    return messages[key];
  }
}
```
